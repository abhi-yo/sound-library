## Integration Guide

<Steps>
  <Step>
    ### Import the Audio Trigger
    Import `playSound` or paste the standalone function into your component:

    ```ts title="lib/sounds.ts"
    import { playSound } from '@/lib/sounds';
    ```

  </Step>

  <Step>
    ### Wire to Interactions
    Trigger on rewarding actions, level-ups, or playful UI elements:

    ```tsx title="components/reward-button.tsx"
    'use client';

        <button
    ```

### 8-bit Coin (`coin`)

<SoundPlayButton name="coin" />

Classic 8-bit coin collection ping for gamified rewards.

```ts title="lib/sounds/coin.ts"
export function playCoin() {
  const ctx = new AudioContext();
  const now = ctx.currentTime;
  const osc = ctx.createOscillator();
  const gain = ctx.createGain();
  osc.type = 'square';
  osc.frequency.setValueAtTime(988, now); // B5
  osc.frequency.setValueAtTime(1318.5, now + 0.08); // E6
  gain.gain.setValueAtTime(0.1, now);
  gain.gain.setValueAtTime(0.1, now + 0.08);
  gain.gain.exponentialRampToValueAtTime(0.001, now + 0.3);
  osc.connect(gain).connect(ctx.destination);
  osc.start(now);
  osc.stop(now + 0.35);
}
```

```ts title="lib/sounds/power-up.ts"
export function playPowerUp() {
```

### Typewriter Key (`typewriter-key`)

<SoundPlayButton name="typewriter-key" />

Crisp, vintage mechanical typewriter clack for text reveals.

```ts title="lib/sounds/typewriter-key.ts"
export function playTypewriterKey() {
  const ctx = new AudioContext();
  const now = ctx.currentTime;
  const bufSize = Math.floor(ctx.sampleRate * 0.04);
  const buffer = ctx.createBuffer(1, bufSize, ctx.sampleRate);
  const data = buffer.getChannelData(0);
  for (let i = 0; i < bufSize; i++) data[i] = Math.random() * 2 - 1;
  const noise = ctx.createBufferSource();
  noise.buffer = buffer;
  const bp = ctx.createBiquadFilter();
  bp.type = 'bandpass';
  bp.frequency.setValueAtTime(2500, now);
  bp.Q.setValueAtTime(1.5, now);
  const gain = ctx.createGain();
  gain.gain.setValueAtTime(0.25, now);
  gain.gain.exponentialRampToValueAtTime(0.001, now + 0.03);
  noise.connect(bp).connect(gain).connect(ctx.destination);
  noise.start(now);
  noise.stop(now + 0.04);
}
```

### Digital Glitch (`glitch`)

<SoundPlayButton name="glitch" />

Fast, erratic, randomized noise and square waves for error or cyber themes.

```ts title="lib/sounds/glitch.ts"
export function playGlitch() {
  const ctx = new AudioContext();
  const now = ctx.currentTime;
  
  // 1. Harsh Sawtooth Buzz ("bzzzt")
  const buzz = ctx.createOscillator();
  buzz.type = 'sawtooth';
  // Frequency jumps to simulate digital corruption
  buzz.frequency.setValueAtTime(60, now);
  buzz.frequency.setValueAtTime(120, now + 0.06);
  
  const buzzGain = ctx.createGain();
  buzzGain.gain.setValueAtTime(0.3, now);
  // "chk" stutter
  buzzGain.gain.setValueAtTime(0, now + 0.02);
  buzzGain.gain.setValueAtTime(0.3, now + 0.03);
  buzzGain.gain.setValueAtTime(0, now + 0.05);
  // main buzz
  buzzGain.gain.setValueAtTime(0.3, now + 0.06);
  buzzGain.gain.exponentialRampToValueAtTime(0.001, now + 0.12);
  
  buzz.connect(buzzGain).connect(ctx.destination);
  buzz.start(now);
  buzz.stop(now + 0.13);

  // 2. Digital Static ("krrrk")
  const bufSize = Math.floor(ctx.sampleRate * 0.15);
  const buffer = ctx.createBuffer(1, bufSize, ctx.sampleRate);
  const data = buffer.getChannelData(0);
  for (let i = 0; i < bufSize; i++) data[i] = Math.random() * 2 - 1;
  const noise = ctx.createBufferSource();
  noise.buffer = buffer;
  
  // Highpass to make it thin and digital
  const hp = ctx.createBiquadFilter();
  hp.type = 'highpass';
  hp.frequency.setValueAtTime(3000, now);
  
  const noiseGain = ctx.createGain();
  // Sync the stutters with the buzz
  noiseGain.gain.setValueAtTime(0.4, now);
  noiseGain.gain.setValueAtTime(0, now + 0.02);
  noiseGain.gain.setValueAtTime(0.4, now + 0.03);
  noiseGain.gain.setValueAtTime(0, now + 0.05);
  noiseGain.gain.setValueAtTime(0.2, now + 0.06);
  noiseGain.gain.exponentialRampToValueAtTime(0.001, now + 0.12);
  
  noise.connect(hp).connect(noiseGain).connect(ctx.destination);
  noise.start(now);
  noise.stop(now + 0.13);
}
```

## Sound Catalog

### Ascending Chime (`chime`)

<SoundPlayButton name="chime" />

Harmonic two-note ascending interval for saves, successful operations, and completed tasks.

```ts title="lib/sounds/chime.ts"
export function playChime() {
  const ctx = new AudioContext();
  const now = ctx.currentTime;
  
  const n1 = ctx.createOscillator();
  const g1 = ctx.createGain();
  n1.type = 'sine';
  n1.frequency.setValueAtTime(660, now);
  g1.gain.setValueAtTime(0.2, now);
  g1.gain.exponentialRampToValueAtTime(0.001, now + 0.14);
  n1.connect(g1).connect(ctx.destination);
  n1.start(now);
  n1.stop(now + 0.14);

  const n2 = ctx.createOscillator();
  const g2 = ctx.createGain();
  n2.type = 'sine';
  n2.frequency.setValueAtTime(990, now + 0.09);
  g2.gain.setValueAtTime(0.001, now);
  g2.gain.setValueAtTime(0.22, now + 0.09);
  g2.gain.exponentialRampToValueAtTime(0.001, now + 0.28);
  n2.connect(g2).connect(ctx.destination);
  n2.start(now + 0.09);
  n2.stop(now + 0.28);
}
```

### Triad Bell (`bell`)

<SoundPlayButton name="bell" />

Acoustic three-note major arpeggio for achievements, celebration milestones, and confetti bursts.

```ts title="lib/sounds/bell.ts"
export function playBell() {
  const ctx = new AudioContext();
  const now = ctx.currentTime;
  
  [880, 1100, 1320].forEach((freq, i) => {
    const osc = ctx.createOscillator();
    const gain = ctx.createGain();
    osc.type = 'sine';
    osc.frequency.setValueAtTime(freq, now + i * 0.07);
    gain.gain.setValueAtTime(0.001, now);
    gain.gain.setValueAtTime(0.16, now + i * 0.07);
    gain.gain.exponentialRampToValueAtTime(0.001, now + i * 0.07 + 0.18);
    osc.connect(gain).connect(ctx.destination);
    osc.start(now + i * 0.07);
    osc.stop(now + i * 0.07 + 0.18);
  });
}
```

### Magical Shimmer (`shimmer`)

<SoundPlayButton name="shimmer" />

A mystical, multi-layered chord shimmer for highlighting special features, achievements, and rewards.

```ts title="lib/sounds/shimmer.ts"
export function playShimmer() {
  const ctx = new AudioContext();
  const now = ctx.currentTime;
  const freqs = [1200, 1600, 2100, 2800];
  freqs.forEach((freq, idx) => {
    const osc = ctx.createOscillator();
    const gain = ctx.createGain();
    osc.type = 'triangle';
    osc.frequency.setValueAtTime(freq, now + idx * 0.04);
    gain.gain.setValueAtTime(0.001, now);
    gain.gain.setValueAtTime(0.06, now + idx * 0.04);
    gain.gain.exponentialRampToValueAtTime(0.001, now + idx * 0.04 + 0.2);
    osc.connect(gain).connect(ctx.destination);
    osc.start(now + idx * 0.04);
    osc.stop(now + idx * 0.04 + 0.25);

}
```

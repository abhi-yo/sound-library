<Steps>
  <Step>
    ### Import the Audio Trigger
    Import `playSound` or paste the standalone function into your navigation component:

    ```ts title="lib/sounds.ts"
    import { playSound } from '@/lib/sounds';
    ```
  </Step>

  <Step>
    ### Wire to Sheet, Drawer, or Page Routing Handlers
    Trigger on drawer open or layout animation transitions:

    ```tsx title="components/drawer.tsx"
    'use client';

    import { playSound } from '@/lib/sounds';

    export function DrawerToggle({ isOpen, onToggle }: { isOpen: boolean; onToggle: () => void }) {
      return (
        <button
          onClick={() => {
            playSound('swish');
            onToggle();
          }}
          className="rounded-xl border border-border px-3.5 py-1.5 text-xs font-medium"
        >
          {isOpen ? 'Close Panel' : 'Open Panel'}
        </button>
      );
    }
    ```

## Sound Catalog

### Filtered Swish (`swish`)

<SoundPlayButton name="swish" />

Filtered noise sweep with lowpass frequency drop for sheet reveals, modal drawers, and tab switches.

```ts title="lib/sounds/swish.ts"
export function playSwish() {
  const ctx = new AudioContext();
  const now = ctx.currentTime;
  
  const bufferSize = ctx.sampleRate * 0.2;
  const buffer = ctx.createBuffer(1, bufferSize, ctx.sampleRate);
  const data = buffer.getChannelData(0);
  for (let i = 0; i < bufferSize; i++) {
    data[i] = Math.random() * 2 - 1;
  }
  
  const noise = ctx.createBufferSource();
  noise.buffer = buffer;
  
  const filter = ctx.createBiquadFilter();
  filter.type = 'lowpass';
  filter.frequency.setValueAtTime(3200, now);
  filter.frequency.exponentialRampToValueAtTime(350, now + 0.18);
  
  const gain = ctx.createGain();
  gain.gain.setValueAtTime(0.001, now);
  gain.gain.linearRampToValueAtTime(0.18, now + 0.04);
  gain.gain.exponentialRampToValueAtTime(0.001, now + 0.2);
  
  noise.connect(filter).connect(gain).connect(ctx.destination);
  noise.start(now);
  noise.stop(now + 0.2);
}
```

```ts title="lib/sounds/glissando.ts"
export function playGlissando() {
```

### Heavy Woosh (`woosh-deep`)

<SoundPlayButton name="woosh-deep" />

A cinematic, filtered noise sweep for grand page transitions and heavy spatial reveals.

```ts title="lib/sounds/woosh-deep.ts"
export function playWooshDeep() {
  const ctx = new AudioContext();
  const now = ctx.currentTime;
  const bufSize = Math.floor(ctx.sampleRate * 0.4);
  const buffer = ctx.createBuffer(1, bufSize, ctx.sampleRate);
  const data = buffer.getChannelData(0);
  for (let i = 0; i < bufSize; i++) data[i] = Math.random() * 2 - 1;
  const noise = ctx.createBufferSource();
  noise.buffer = buffer;
  const filter = ctx.createBiquadFilter();
  filter.type = 'bandpass';
  filter.frequency.setValueAtTime(100, now);
  filter.frequency.exponentialRampToValueAtTime(800, now + 0.2);
  filter.frequency.exponentialRampToValueAtTime(100, now + 0.4);
  const gain = ctx.createGain();
  gain.gain.setValueAtTime(0.001, now);
  gain.gain.linearRampToValueAtTime(0.2, now + 0.2);
  gain.gain.exponentialRampToValueAtTime(0.001, now + 0.4);
  noise.connect(filter).connect(gain).connect(ctx.destination);
  noise.start(now);
  noise.stop(now + 0.45);
}
```

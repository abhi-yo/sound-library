## Quick Start

Copy any standalone sound function directly into your project:

```ts title="lib/sounds/tap.ts"
export function playTap() {
  const ctx = new AudioContext();
  const now = ctx.currentTime;
  const osc = ctx.createOscillator();
  const gain = ctx.createGain();
  
  osc.type = 'sine';
  osc.frequency.setValueAtTime(1800, now);
  osc.frequency.exponentialRampToValueAtTime(700, now + 0.035);
  
  gain.gain.setValueAtTime(0.18, now);
  gain.gain.exponentialRampToValueAtTime(0.001, now + 0.035);
  
  osc.connect(gain).connect(ctx.destination);
  osc.start(now);
  osc.stop(now + 0.04);
}
```

Trigger it in any component:

```tsx title="components/button.tsx"
'use client';

import { playTap } from '@/lib/sounds/tap';

export function ActionButton() {
  return (
    <button onClick={playTap} className="btn">
      Confirm Action
    </button>
  );
}
```
